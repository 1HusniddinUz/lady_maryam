# Banner
function Show-Banner {
    Clear-Host
    Write-Host ""
    Write-Host "  ╔═══════════════════════════════════════════════════════╗" -ForegroundColor Magenta
    Write-Host "  ║                                                       ║" -ForegroundColor Magenta
    Write-Host "  ║     " -NoNewline -ForegroundColor Magenta
    Write-Host "✦  L A D Y   M A R Y A M   A T E L I E R" -NoNewline -ForegroundColor White
    Write-Host "      ║" -ForegroundColor Magenta
    Write-Host "  ║     " -NoNewline -ForegroundColor Magenta
    Write-Host "Universal Setup Skripti" -NoNewline -ForegroundColor DarkYellow
    Write-Host "                       ║" -ForegroundColor Magenta
    Write-Host "  ║                                                       ║" -ForegroundColor Magenta
    Write-Host "  ╚═══════════════════════════════════════════════════════╝" -ForegroundColor Magenta
    Write-Host ""
}

function Write-Step {
    param([string]$Num, [string]$Title)
    Write-Host ""
    Write-Host "  ┌─────────────────────────────────────────────────────┐" -ForegroundColor DarkCyan
    Write-Host "  │ " -NoNewline -ForegroundColor DarkCyan
    Write-Host " $Num " -NoNewline -ForegroundColor White -BackgroundColor DarkBlue
    Write-Host "  $Title" -NoNewline -ForegroundColor Cyan
    $padding = 50 - $Title.Length - $Num.Length
    Write-Host (" " * $padding) -NoNewline
    Write-Host "│" -ForegroundColor DarkCyan
    Write-Host "  └─────────────────────────────────────────────────────┘" -ForegroundColor DarkCyan
}

function Write-Ok   { param([string]$m) Write-Host "    ✓ " -NoNewline -ForegroundColor Green; Write-Host $m -ForegroundColor White }
function Write-Warn { param([string]$m) Write-Host "    ⚠ " -NoNewline -ForegroundColor Yellow; Write-Host $m -ForegroundColor Yellow }
function Write-Err  { param([string]$m) Write-Host "    ✗ " -NoNewline -ForegroundColor Red; Write-Host $m -ForegroundColor Red }
function Write-Info { param([string]$m) Write-Host "    → " -NoNewline -ForegroundColor DarkGray; Write-Host $m -ForegroundColor Gray }
function Write-Cmd  { param([string]$m) Write-Host "      $m" -ForegroundColor DarkCyan }

function Test-Cmd { param([string]$c) return [bool](Get-Command $c -ErrorAction SilentlyContinue) }

function Ask-Yes {
    param([string]$Question, [bool]$Default = $true)
    Write-Host ""
    $suffix = if ($Default) { "[Y/n]" } else { "[y/N]" }
    Write-Host "    ❓ $Question $suffix " -NoNewline -ForegroundColor Yellow
    $answer = Read-Host
    if (-not $answer) { return $Default }
    return $answer -match "^[yY]"
}

function Wait-Enter {
    param([string]$Msg = "Davom etish uchun Enter bosing...")
    Write-Host ""
    Write-Host "    $Msg" -ForegroundColor DarkGray
    [void](Read-Host)
}

$hasLocalChanges = $false

if (Test-Path $oldPath) {
    Write-Info "ZIP papka topildi: $oldPath"

    if (Test-Path "$oldPath\.git") {
        Write-Ok "Bu git repo (clone qilingan)"
        if (Test-Path $newPath) {
            Write-Warn "$($CONFIG.NewFolder) ham mavjud"
            if (Ask-Yes "Eski $($CONFIG.OldFolder)'ni o'chirib yuborishni xohlaysizmi?") {
                Remove-Item $oldPath -Recurse -Force
                Write-Ok "Eski papka o'chirildi"
            }
        } else {
            # rename qilamiz
            Rename-Item $oldPath $CONFIG.NewFolder
            Write-Ok "$($CONFIG.OldFolder) → $($CONFIG.NewFolder) rename qilindi"
        }
    } else {
        Write-Warn "Bu ZIP archive — .git papka yo'q"
        Write-Info "Sizning kodingizni zaxiralayman..."

        if (Test-Path $backupPath) {
            $oldBackupName = "$($CONFIG.BackupFolder)-$(Get-Date -Format 'yyyyMMdd_HHmm')"
            Rename-Item $backupPath $oldBackupName
            Write-Info "Eski backup saqlandi: $oldBackupName"
        }

        Copy-Item $oldPath $backupPath -Recurse -Force
        Write-Ok "Zaxira yaratildi: $backupPath"
        $hasLocalChanges = $true

        Remove-Item $oldPath -Recurse -Force
        Write-Ok "ZIP papka o'chirildi"
    }
} else {
    Write-Info "$($CONFIG.OldFolder) topilmadi"
    if (Test-Path $newPath -and (Test-Path "$newPath\.git")) {
        Write-Ok "$($CONFIG.NewFolder) mavjud va git repo"
    } else {
        Write-Info "Toza clone qilamiz"
    }
}

            Write-Warn "GitHub'da $behind ta yangi commit bor"

                git pull origin $CONFIG.Branch

} else {

    try {
        $cloneResult = git clone $CONFIG.GitHubUrl $CONFIG.NewFolder 2>&1
        $cloneResult | ForEach-Object { Write-Host "      $_" -ForegroundColor DarkGray }

        if (-not (Test-Path $newPath)) {
            throw "Clone amalga oshmadi"
        }
        Write-Ok "Clone muvaffaqiyatli"
        Set-Location $newPath
    } catch {
        Write-Err "Clone xatosi: $_"
        Write-Host ""
        Write-Info "Mumkin bo'lgan sabablar:"
        Write-Info "  1. Repo PRIVATE — Personal Access Token kerak"
        Write-Info "  2. Internet muammosi"

}

# 4️⃣  LOCAL O'ZGARISHLARNI QAYTARISH
# ════════════════════════════════════════════════════════════════

Write-Step "4/10" "Local o'zgarishlarni birlashtirish"

if ($hasLocalChanges -and (Test-Path $backupPath)) {
    if (Ask-Yes "Zaxiradagi kodlarni hozirgi repo'ga ko'chiramizmi?") {
        $excludeDirs = @("venv", "node_modules", "__pycache__", ".git", "dist", "build", ".idea", ".vscode", ".pytest_cache")
        $excludeFiles = @("*.pyc", "*.pyo", "*.log", "*.tmp")

        $copied = 0
        $skipped = 0

        Get-ChildItem -Path $backupPath -Recurse -File -ErrorAction SilentlyContinue | ForEach-Object {
            $relPath = $_.FullName.Substring($backupPath.Length + 1)

            $skip = $false
            foreach ($dir in $excludeDirs) {
                if ($relPath -match "(^|[\\/])$dir([\\/]|$)") { $skip = $true; break }
            }
            if (-not $skip) {
                foreach ($pattern in $excludeFiles) {
                    if ($_.Name -like $pattern) { $skip = $true; break }
                }
            }

            if ($skip) { $skipped++; return }

            $destPath = Join-Path $newPath $relPath
            $destDir = Split-Path $destPath -Parent
            if (-not (Test-Path $destDir)) {
                New-Item -ItemType Directory -Path $destDir -Force | Out-Null
            }

            $shouldCopy = $true
            if (Test-Path $destPath) {
                try {
                    $oldHash = (Get-FileHash $destPath -Algorithm MD5).Hash
                    $newHash = (Get-FileHash $_.FullName -Algorithm MD5).Hash
                    if ($oldHash -eq $newHash) { $shouldCopy = $false }
                } catch { }
            }

            if ($shouldCopy) {
                Copy-Item -Path $_.FullName -Destination $destPath -Force
                $copied++
            }
        }

        Write-Ok "$copied ta fayl ko'chirildi, $skipped ta o'tkazib yuborildi"

        # Git status
        Write-Info "O'zgarishlar:"
        $changes = git status --porcelain 2>$null
        if ($changes) {
            $changes | Select-Object -First 8 | ForEach-Object {
                Write-Host "      $_" -ForegroundColor Yellow
            }
            if ($changes.Count -gt 8) {
                Write-Info "... yana $($changes.Count - 8) ta"
            }
        } else {
            Write-Info "O'zgarishlar yo'q (kodingiz GitHub bilan bir xil edi)"
        }
    } else {
        Write-Info "Zaxira o'z joyida qoldi: $backupPath"
    }
} else {
    Write-Info "Local o'zgarishlar yo'q (yoki zaxira yo'q)"
}

# .gitignore
$gitignoreNeeded = @(".env", "venv/", "__pycache__/", "*.pyc", ".vscode/", "node_modules/", ".DS_Store", "backup_*.sql", "*.dump")
$gitignoreContent = ""
if (Test-Path ".gitignore") {
    $gitignoreContent = Get-Content ".gitignore" -Raw
}

$additions = @()
foreach ($entry in $gitignoreNeeded) {
    $escapedEntry = [regex]::Escape($entry)
    if ($gitignoreContent -notmatch "(?m)^$escapedEntry`?$") {
        $additions += $entry
    }
}

if ($additions.Count -gt 0) {
    Add-Content -Path ".gitignore" -Value "`n# Auto-added`n$($additions -join "`n")`n"
    Write-Ok ".gitignore yangilandi: $($additions.Count) ta qoida qo'shildi"
} else {
    Write-Ok ".gitignore to'g'ri sozlangan"
}

## Database schema

### `debts` jadvali

```sql
CREATE TABLE IF NOT EXISTS debts (
    id SERIAL PRIMARY KEY,
    user_id BIGINT NOT NULL,                 -- admin telegram_id
    customer_name TEXT NOT NULL,
    customer_phone TEXT NOT NULL,
    customer_telegram TEXT,                  -- username (without @)
    customer_chat_id BIGINT,                 -- mijoz chat_id (eslatma uchun)
    items JSONB NOT NULL,                    -- [{name, qty, price}]
    total_amount BIGINT NOT NULL,
    paid_amount BIGINT DEFAULT 0,
    taken_date DATE NOT NULL,
    due_date DATE NOT NULL,
    notes TEXT,
    status TEXT DEFAULT 'active',            -- active | paid
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_debts_due_date ON debts(due_date);
CREATE INDEX IF NOT EXISTS idx_debts_status ON debts(status);
CREATE INDEX IF NOT EXISTS idx_debts_customer_chat ON debts(customer_chat_id);
```

### `debt_reminders` jadvali

```sql
CREATE TABLE IF NOT EXISTS debt_reminders (
    id SERIAL PRIMARY KEY,
    debt_id INTEGER REFERENCES debts(id) ON DELETE CASCADE,
    sent_at TIMESTAMP DEFAULT NOW(),
    tone TEXT,                               -- polite | urgent | friendly
    message TEXT NOT NULL,
    delivery_status TEXT,                    -- sent | failed | manual
    channel TEXT DEFAULT 'telegram'
);

CREATE INDEX IF NOT EXISTS idx_reminders_debt ON debt_reminders(debt_id);
```

```python
# bot/scheduler.py

from apscheduler.schedulers.asyncio import AsyncIOScheduler
from apscheduler.triggers.cron import CronTrigger
import pytz

scheduler = AsyncIOScheduler(timezone=pytz.timezone("Asia/Tashkent"))

@scheduler.scheduled_job(CronTrigger(hour=9, minute=0))
async def check_due_debts():
    today = date.today()

    due_debts = await db.fetch_all("""
        SELECT * FROM debts
        WHERE due_date <= $1
          AND paid_amount < total_amount
          AND id NOT IN (
              SELECT debt_id FROM debt_reminders
              WHERE DATE(sent_at) = CURRENT_DATE
          )
    """, today)

    for debt in due_debts:
        await send_debt_reminder(debt)
```

### Eslatma yuborish

```python
async def send_debt_reminder(debt: dict):
    days_overdue = (date.today() - debt["due_date"]).days
    tone = "urgent" if days_overdue > 0 else "polite"
    message = await generate_reminder_text(debt, tone)

    if debt["customer_chat_id"]:
        try:
            await bot.send_message(debt["customer_chat_id"], message)
            await log_reminder(debt["id"], tone, message, "sent")
        except (TelegramForbiddenError, TelegramBadRequest):
            await log_reminder(debt["id"], tone, message, "failed")
            await notify_admin_failed(debt, message)
    else:
        await notify_admin_manual(debt, message)
        await log_reminder(debt["id"], tone, message, "manual")
```

```python
async def generate_reminder_text(debt: dict, tone: str = "polite") -> str:
    days = (debt["due_date"] - date.today()).days
    day_context = (
        f"Muddat {abs(days)} kun avval o'tgan" if days < 0
        else "Muddat bugun" if days == 0
        else f"Muddat {days} kun qoldi"
    )

    tone_desc = {
        "urgent": "qattiqroq, lekin hurmatli",
        "friendly": "juda do'stona va yumshoq",
        "polite": "rasmiy, hurmatli, lekin aniq"
    }[tone]

    prompt = f"""Lady Maryam Atelier mijoziga qarzni qaytarish bo'yicha eslatma yoz.

Mijoz: {debt['customer_name']}
Qarz: {debt['total_amount'] - debt['paid_amount']:,} so'm
Muddat: {debt['due_date']}
Holat: {day_context}
Ohang: {tone_desc}

TALABLAR:
- O'ZBEK tilida, 3-4 jumla
- "Hurmatli {debt['customer_name']}" deb murojaat
- Aniq miqdor va muddat
- Telegram uchun qisqa
- Imzo: "Lady Maryam Atelier"
- Faqat xabar matni"""

    client = anthropic.AsyncAnthropic(api_key=settings.ANTHROPIC_API_KEY)
    response = await client.messages.create(
        model="claude-sonnet-4-20250514",
        max_tokens=500,
        messages=[{"role": "user", "content": prompt}]
    )
    return response.content[0].text.strip()
```
